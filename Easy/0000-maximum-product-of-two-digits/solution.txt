
        while (n > 0) {
            int digit = n % 10;
            
            if (digit > max1) {
                max2 = max1;
                max1 = digit;
            } else {
                max2 = Math.max(max2, digit);
            }

            n /= 10;
        }

        return max1 * max2;
    }
}
