
    // Computes nCr capped at limit
    private long comb(int n, int r, long limit) {
        if (r < 0 || r > n) return 0;
        if (r == 0 || r == n) return 1;
        if (r > n - r) r = n - r;

        long res = 1;
        for (int i = 1; i <= r; i++) {
            res = res * (n - i + 1) / i;
            if (res > limit) {
                return limit + 1;
            }
        }
        return res;
    }
}
